    57→// Shared fetch helper for mutations (CSRF + JSON)
    58→async function adminFetch(url: string, method: string, body: unknown): Promise<Response> {
    59→  const csrfMatch = document.cookie.split("; ").find((c) => c.startsWith("futsal_csrf="));
    60→  const csrfToken = csrfMatch?.split("=")[1] ?? "";
    61→  return fetch(url, {
    62→    method,
    63→    credentials: "same-origin",
    64→    headers: { "Content-Type": "application/json", [CSRF_HEADER]: csrfToken },
    65→    body: JSON.stringify(body),
    66→  });
    67→}

   247→// === Student Management ==================================================

   249→type ListState =
   250→  | { status: "loading" }
   251→  | { status: "success"; data: StudentRow[] }
   252→  | { status: "error"; message: string };
   253→
   254→function StudentManagement({ academyId }: { academyId: string }) {
   255→  const [state, setState] = useState<ListState>({ status: "loading" });
   256→  const [searchQuery, setSearchQuery] = useState("");
   257→  const [selectedStudentId, setSelectedStudentId] = useState<string | null>(null);
   258→  const [showCreateDialog, setShowCreateDialog] = useState(false);
   259→
   260→  const academyIdRef = useRef(academyId);
   261→  useEffect(() => { academyIdRef.current = academyId; }, [academyId]);
   262→
   263→  const fetchList = useCallback(async () => {
   264→    const aid = academyId;
   265→    setState({ status: "loading" });
   266→    try {
   267→      const res = await fetch(`/api/admin/students?academyId=${aid}`, { credentials: "same-origin" });
   268→      const data = (await res.json()) as { students?: StudentRow[]; error?: { message?: string } };
   269→      if (aid !== academyIdRef.current) return;
   270→      if (!res.ok || !data.students) {
   271→        setState({ status: "error", message: data.error?.message ?? `Error ${res.status}` });
   272→        return;
   273→      }
   274→      setState({ status: "success", data: data.students });
   275→    } catch {
   276→      if (aid !== academyIdRef.current) return;
   277→      setState({ status: "error", message: "Network error" });
   278→    }
   279→  }, [academyId]);
   280→
   281→  useEffect(() => {
   282→    if (!academyId) return;
   283→    let cancelled = false;
   284→    const aid = academyId;
   285→    void (async () => {
   286→      try {
   287→        const res = await fetch(`/api/admin/students?academyId=${aid}`, { credentials: "same-origin" });
   288→        if (cancelled) return;
   289→        const data = (await res.json()) as { students?: StudentRow[]; error?: { message?: string } };
   290→        if (cancelled) return;
   291→        if (!res.ok || !data.students) {
   292→          setState({ status: "error", message: data.error?.message ?? `Error ${res.status}` });
   293→          return;
   294→        }
   295→        setState({ status: "success", data: data.students });
   296→      } catch {
   297→        if (cancelled) return;
   298→        setState({ status: "error", message: "Network error" });
   299→      }
   300→    })();
   301→    return () => { cancelled = true; };
   302→  }, [academyId]);
   303→
   304→  // Client-side search filtering (API doesn't support search params)
   305→  const filteredStudents = useMemo(() => {
   306→    if (state.status !== "success") return [];
   307→    const q = searchQuery.toLowerCase().trim();
   308→    if (!q) return state.data;
   309→    return state.data.filter((s) =>
   310→      s.full_name.toLowerCase().includes(q)
   311→    );
   312→  }, [state, searchQuery]);
   313→
   314→  // Student detail view
   315→  if (selectedStudentId) {
   316→    return (
   317→      <StudentDetail
   318→        studentId={selectedStudentId}
   319→        academyId={academyId}
   320→        onBack={() => { setSelectedStudentId(null); fetchList(); }}
   321→      />
   322→    );
   323→  }
   324→
   325→  return (
   326→    <div className="space-y-4">
   327→      {/* Toolbar */}
   328→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   329→        <div className="relative max-w-xs flex-1">
   330→          <Search className="absolute left-2 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   331→          <Input
   332→            placeholder="Search students…"
   333→            value={searchQuery}
   334→            onChange={(e) => setSearchQuery(e.target.value)}
   335→            className="pl-8"
   336→            disabled={state.status === "loading"}
   337→          />
   338→        </div>
   339→        <Button className="gap-2" onClick={() => setShowCreateDialog(true)}>
   340→          <UserPlus className="h-4 w-4" aria-hidden /> Add Student
   341→        </Button>
   342→      </div>
   343→
   344→      {/* States */}
   345→      {state.status === "loading" && (
   346→        <Card>
   347→          <CardContent className="flex items-center gap-3 p-6">
   348→            <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" aria-hidden />
   349→            <p className="text-sm text-muted-foreground">Loading students…</p>
   350→          </CardContent>
   351→        </Card>
   352→      )}
   353→
   354→      {state.status === "error" && (
   355→        <Card className="border-destructive/30">
   356→          <CardContent className="p-6">
   357→            <div className="flex items-start gap-3">
   358→              <AlertCircle className="h-5 w-5 text-destructive flex-shrink-0 mt-0.5" aria-hidden />
   359→              <div>
   360→                <p className="text-sm font-medium text-destructive">Failed to load students</p>
   361→                <p className="mt-1 text-sm text-muted-foreground">{state.message}</p>
   362→                <Button variant="outline" size="sm" className="mt-3 gap-2" onClick={fetchList}>
   363→                  <RefreshCw className="h-4 w-4" aria-hidden /> Retry
   364→                </Button>
   365→              </div>
   366→            </div>
   367→          </CardContent>
   368→        </Card>
   369→      )}
   370→
   371→      {state.status === "success" && (
   372→        <>
   373→          {filteredStudents.length === 0 ? (
   374→            <Card>
   375→              <CardContent className="p-8 text-center">
   376→                <Users className="mx-auto h-8 w-8 text-muted-foreground" aria-hidden />
   377→                <p className="mt-3 text-sm font-medium">
   378→                  {searchQuery ? "No students match your search" : "No students"}
   379→                </p>
   380→                <p className="mt-1 text-sm text-muted-foreground">
   381→                  {searchQuery
   382→                    ? "Try a different search term."
   383→                    : "Click 'Add Student' to create the first student."}
   384→                </p>
   385→              </CardContent>
   386→            </Card>
   387→          ) : (
   388→            <>
   389→              <Badge variant="secondary">
   390→                {filteredStudents.length} student{filteredStudents.length !== 1 ? "s" : ""}
   391→              </Badge>
   392→              {/* Desktop: table */}
   393→              <Card>
   394→                <CardContent className="p-0">
   395→                  <div className="hidden sm:block">
   396→                    <Table>
   397→                      <TableHeader>
   398→                        <TableRow>
   399→                          <TableHead>Name</TableHead>
   400→                          <TableHead>Date of Birth</TableHead>
   401→                          <TableHead>Gender</TableHead>
   402→                          <TableHead className="hidden md:table-cell">Notes</TableHead>
   403→                        </TableRow>
   404→                      </TableHeader>
   405→                      <TableBody>
   406→                        {filteredStudents.map((s) => (
   407→                          <TableRow
   408→                            key={s.id}
   409→                            className="cursor-pointer hover:bg-muted/50"
   410→                            onClick={() => setSelectedStudentId(s.id)}
   411→                          >
   412→                            <TableCell className="font-medium">{s.full_name}</TableCell>
   413→                            <TableCell className="tabular-nums">{s.date_of_birth ?? "—"}</TableCell>
   414→                            <TableCell>{s.gender ?? "—"}</TableCell>
   415→                            <TableCell className="max-w-xs truncate text-muted-foreground hidden md:table-cell">
   416→                              {s.notes ?? "—"}
   417→                            </TableCell>
   418→                          </TableRow>
   419→                        ))}
   420→                      </TableBody>
   421→                    </Table>
   422→                  </div>
   423→                  {/* Mobile: cards */}
   424→                  <div className="divide-y sm:hidden">
   425→                    {filteredStudents.map((s) => (
   426→                      <button
   427→                        key={s.id}
   428→                        type="button"
   429→                        className="w-full p-4 text-left hover:bg-muted/50"
   430→                        onClick={() => setSelectedStudentId(s.id)}
   431→                      >
   432→                        <p className="font-medium">{s.full_name}</p>
   433→                        <div className="mt-1 flex flex-wrap gap-x-3 gap-y-0.5 text-sm text-muted-foreground">
   434→                          {s.date_of_birth && <span className="tabular-nums">{s.date_of_birth}</span>}
   435→                          {s.gender && <span>{s.gender}</span>}
   436→                        </div>
   437→                      </button>
   438→                    ))}
   439→                  </div>
   440→                </CardContent>
   441→              </Card>
   442→            </>
   443→          )}
   444→        </>
   445→      )}
   446→
   447→      {/* Create student dialog */}
   448→      <CreateStudentDialog
   449→        open={showCreateDialog}
   450→        onClose={() => setShowCreateDialog(false)}
   451→        academyId={academyId}
   452→        onCreated={() => { setShowCreateDialog(false); fetchList(); }}
   453→      />
   454→    </div>
   455→  );
   456→}

   458→// === Create Student Dialog ===============================================
   459→
   460→function CreateStudentDialog({
   461→  open,
   462→  onClose,
   463→  academyId,
   464→  onCreated,
   465→}: {
   466→  open: boolean;
   467→  onClose: () => void;
   468→  academyId: string;
   469→  onCreated: () => void;
   470→}) {
   471→  const [fullName, setFullName] = useState("");
   472→  const [dateOfBirth, setDateOfBirth] = useState("");
   473→  const [gender, setGender] = useState("");
   474→  const [notes, setNotes] = useState("");
   475→  const [submitting, setSubmitting] = useState(false);
   476→  const [error, setError] = useState<string | null>(null);
   477→
   478→  useEffect(() => {
   479→    if (open) {
   480→      setFullName(""); setDateOfBirth(""); setGender(""); setNotes("");
   481→      setSubmitting(false); setError(null);
   482→    }
   483→  }, [open]);
   484→
   485→  const handleSubmit = async () => {
   486→    if (!fullName.trim()) { setError("Full name is required."); return; }
   487→    setSubmitting(true);
   488→    setError(null);
   489→    try {
   490→      const csrfMatch = document.cookie.split("; ").find((c) => c.startsWith("futsal_csrf="));
   491→      const csrfToken = csrfMatch?.split("=")[1] ?? "";
   492→      const body: Record<string, unknown> = { fullName: fullName.trim() };
   493→      if (dateOfBirth) body.dateOfBirth = dateOfBirth;
   494→      if (gender) body.gender = gender;
   495→      if (notes.trim()) body.notes = notes.trim();
   496→      const res = await fetch(`/api/admin/students?academyId=${academyId}`, {
   497→        method: "POST",
   498→        credentials: "same-origin",
   499→        headers: { "Content-Type": "application/json", [CSRF_HEADER]: csrfToken },
   500→        body: JSON.stringify(body),
   501→      });
   502→      if (res.status === 201) { onCreated(); return; }
   503→      const data = (await res.json()) as { error?: { message?: string } };
   504→      setError(data.error?.message ?? `Error ${res.status}`);
   505→    } catch {
   506→      setError("Network error — could not reach the server.");
   507→    } finally {
   508→      setSubmitting(false);
   509→    }
   510→  };
   511→
   512→  const handleClose = () => {
   513→    if (submitting) return;
   514→    onClose();
   515→  };
   516→
   517→  return (
   518→    <Dialog open={open} onOpenChange={(v) => { if (!v) handleClose(); }}>
   519→      <DialogContent className="sm:max-w-md">
   520→        <DialogHeader>
   521→          <DialogTitle>Add New Student</DialogTitle>
   522→          <DialogDescription>Create a new student in this academy.</DialogDescription>
   523→        </DialogHeader>
   524→        <div className="space-y-4">
   525→          <div className="space-y-2">
   526→            <Label htmlFor="create-fullName">Full Name *</Label>
   527→            <Input id="create-fullName" value={fullName} onChange={(e) => setFullName(e.target.value)} disabled={submitting} placeholder="e.g., Budi Santoso" />
   528→          </div>
   529→          <div className="space-y-2">
   530→            <Label htmlFor="create-dob">Date of Birth</Label>
   531→            <Input id="create-dob" type="date" value={dateOfBirth} onChange={(e) => setDateOfBirth(e.target.value)} disabled={submitting} />
   532→          </div>
   533→          <div className="space-y-2">
   534→            <Label htmlFor="create-gender">Gender</Label>
   535→            <Select value={gender} onValueChange={setGender} disabled={submitting}>
   536→              <SelectTrigger id="create-gender"><SelectValue placeholder="Select gender" /></SelectTrigger>
   537→              <SelectContent>
   538→                <SelectItem value="male">Male</SelectItem>
   539→                <SelectItem value="female">Female</SelectItem>
   540→                <SelectItem value="unspecified">Unspecified</SelectItem>
   541→              </SelectContent>
   542→            </Select>
   543→          </div>
   544→          <div className="space-y-2">
   545→            <Label htmlFor="create-notes">Notes</Label>
   546→            <Textarea id="create-notes" value={notes} onChange={(e) => setNotes(e.target.value)} disabled={submitting} rows={2} placeholder="Optional notes…" />
   547→          </div>
   548→          {error && (
   549→            <div className="flex items-start gap-2 rounded-md border border-destructive/30 bg-destructive/5 p-3">
   550→              <AlertCircle className="h-4 w-4 text-destructive flex-shrink-0 mt-0.5" aria-hidden />

   605→      ]);
   606→      if (cancelled) return;
   607→      if (!studentRes.ok) {
   608→        const err = (await studentRes.json()) as { error?: { message?: string } };
   609→        setState({ status: "error", message: err.error?.message ?? `Error ${studentRes.status}` });
   610→        return;

  1689→function GenerateDialog({ open, onClose, classId, academyId, onGenerated }: {
  1690→  open: boolean; onClose: () => void; classId: string; academyId: string; onGenerated: (msg: string) => void;
  1691→}) {

  1732→        </DialogFooter>
  1733→      </DialogContent>
  1734→    </Dialog>
