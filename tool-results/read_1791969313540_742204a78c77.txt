   121→/** Today's date in Asia/Jakarta (UTC+7), as YYYY-MM-DD. */
   122→function jakartaTodayYmd(): string {
   123→  const shifted = new Date(Date.now() + 7 * 60 * 60 * 1000);
   124→  const y = shifted.getUTCFullYear();
   125→  const m = String(shifted.getUTCMonth() + 1).padStart(2, "0");
   126→  const d = String(shifted.getUTCDate()).padStart(2, "0");
   127→  return `${y}-${m}-${d}`;
   128→}
   129→
   130→/** First day of the current month in Jakarta, YYYY-MM-DD. */
   131→function jakartaMonthStart(): string {
   132→  const shifted = new Date(Date.now() + 7 * 60 * 60 * 1000);
   133→  const y = shifted.getUTCFullYear();
   134→  const m = String(shifted.getUTCMonth() + 1).padStart(2, "0");
   135→  return `${y}-${m}-01`;
   136→}
   137→
   138→/** First day of next month in Jakarta (exclusive upper bound). */
   139→function jakartaMonthEnd(): string {
   140→  const shifted = new Date(Date.now() + 7 * 60 * 60 * 1000);
   141→  let y = shifted.getUTCFullYear();
   142→  let m = shifted.getUTCMonth() + 1; // 1-12
   143→  if (m === 12) {
   144→    m = 1;
   145→    y += 1;
   146→  } else {
   147→    m += 1;
   148→  }
   149→  return `${y}-${String(m).padStart(2, "0")}-01`;
   150→}

   152→/** Format an IDR amount string (e.g. "500000.00") as "Rp 500.000". */
   153→function formatRupiah(amountStr: string): string {
   154→  const n = Number(amountStr);
   155→  if (Number.isNaN(n)) return `Rp ${amountStr}`;
   156→  return `Rp ${n.toLocaleString("id-ID", { maximumFractionDigits: 0 })}`;
   157→}
   158→
   159→/** Format a YYYY-MM-DD as "DD Mon YYYY". */
   160→function formatDateShort(ymd: string): string {
   161→  const d = new Date(`${ymd}T00:00:00.000Z`);
   162→  if (Number.isNaN(d.getTime())) return ymd;
   163→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   164→  return `${String(d.getUTCDate()).padStart(2, "0")} ${months[d.getUTCMonth()]} ${d.getUTCFullYear()}`;
   165→}

   167→async function adminFetch(url: string, method: string, body?: unknown): Promise<Response> {
   168→  const csrfMatch = document.cookie.split("; ").find((c) => c.startsWith("futsal_csrf="));
   169→  const csrfToken = csrfMatch?.split("=")[1] ?? "";
   170→  return fetch(url, {
   171→    method,
   172→    credentials: "same-origin",
   173→    headers: {
   174→      "Content-Type": "application/json",
   175→      [CSRF_HEADER]: csrfToken,
   176→    },
   177→    body: body !== undefined ? JSON.stringify(body) : undefined,
   178→  });
   179→}

   525→function PaymentCard({ payment, academyId }: { payment: AcademyPayment; academyId: string }) {
   526→  const [showMessage, setShowMessage] = useState(false);
   527→  const [message, setMessage] = useState<string | null>(null);
   528→  const [copied, setCopied] = useState(false);
   529→
   530→  const handleShowMessage = async () => {
   531→    if (message) {
   532→      setShowMessage(true);
   533→      return;
   534→    }
   535→    try {
   536→      const res = await fetch(`/api/admin/payments/${payment.id}/message?academyId=${academyId}`, {
   537→        credentials: "same-origin",
   538→      });
   539→      const data = (await res.json()) as { message?: string; error?: { message?: string } };
   540→      if (res.ok && data.message) {
   541→        setMessage(data.message);
   542→        setShowMessage(true);
   543→      }
   544→    } catch {
   545→      // ignore
   546→    }
   547→  };
   548→
   549→  const handleCopy = async () => {
   550→    if (!message) return;
   551→    try {
   552→      await navigator.clipboard.writeText(message);
   553→      setCopied(true);
   554→      setTimeout(() => setCopied(false), 1500);
   555→    } catch {
   556→      // ignore
   557→    }
   558→  };
   559→
   560→  return (
   561→    <Card className="transition-all hover:border-primary/40">
   562→      <CardContent className="p-4">
   563→        <div className="flex items-start justify-between gap-3">
   564→          <div className="flex items-start gap-3">
   565→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-50 text-emerald-600 dark:bg-emerald-950/50 dark:text-emerald-400">
   566→              <ArrowUpCircle className="h-5 w-5" aria-hidden />
   567→            </div>
   568→            <div className="min-w-0">
   569→              <div className="flex flex-wrap items-center gap-2">
   570→                <p className="font-medium leading-tight">{payment.studentName}</p>

   576→                  className={
   577→                    payment.status === "confirmed"
   578→                      ? "border-emerald-200 text-emerald-700 dark:border-emerald-900 dark:text-emerald-400"
   579→                      : payment.status === "submitted"
   580→                        ? "border-amber-200 text-amber-700 dark:border-amber-900 dark:text-amber-400"
   581→                        : "border-muted-foreground/30 text-muted-foreground"
   582→                  }
   583→                >
   584→                  {payment.status}
   585→                </Badge>
   586→              </div>
   587→              <p className="mt-0.5 text-sm text-muted-foreground">
   588→                {payment.packageName ?? "—"}
   589→                {payment.sessionCount ? ` · ${payment.sessionCount} sessions` : ""}
   590→                {payment.payerName ? ` · paid by ${payment.payerName}` : ""}
   591→              </p>
   592→              <p className="mt-0.5 text-xs text-muted-foreground">

   598→          <div className="flex flex-col items-end gap-1.5">

   675→function RecordPaymentDialog({
   676→  open,
   677→  onClose,
   678→  academyId,
   679→  onRecorded,
   680→}: {
   681→  open: boolean;
   682→  onClose: () => void;
   683→  academyId: string;
   684→  onRecorded: () => void;
   685→}) {
   686→  const [students, setStudents] = useState<Student[]>([]);
   687→  const [plans, setPlans] = useState<PackagePlan[]>([]);
   688→  const [studentId, setStudentId] = useState("");
   689→  const [packagePlanId, setPackagePlanId] = useState("");
   690→  const [amount, setAmount] = useState("");
   691→  const [paymentMethod, setPaymentMethod] = useState<"cash" | "bank_transfer">("cash");
   692→  const [paymentDate, setPaymentDate] = useState(jakartaTodayYmd());
   693→  const [payerName, setPayerName] = useState("");
   694→  const [referenceNumber, setReferenceNumber] = useState("");
   695→  const [notes, setNotes] = useState("");
   696→  const [submitting, setSubmitting] = useState(false);
   697→  const [error, setError] = useState<string | null>(null);
   698→  const idempotencyKeyRef = useRef<string>("");
   699→  useEffect(() => {
   700→    idempotencyKeyRef.current = "";
   701→  }, [open]);
   702→
   703→  // Load students + plans when the dialog opens.
   704→  useEffect(() => {
   705→    if (!open) return;
   706→    void (async () => {
   707→      try {
   708→        const [stuRes, planRes] = await Promise.all([
   709→          fetch(`/api/admin/students?academyId=${academyId}`, { credentials: "same-origin" }),
   710→          fetch(`/api/admin/package-plans?academyId=${academyId}`, { credentials: "same-origin" }),
   711→        ]);
   712→        const stuData = (await stuRes.json()) as { students?: Student[] };
   713→        const planData = (await planRes.json()) as { plans?: PackagePlanApi[] };
   714→        setStudents(stuData.students ?? []);
   715→        // Only show active plans in the picker.
   716→        setPlans((planData.plans ?? []).filter((p) => p.is_active).map(mapPlan));
   717→      } catch {
   718→        // ignore — the form will show an empty-list state
   719→      }
   720→    })();
   721→  }, [open, academyId]);
   722→
   723→  // Auto-fill the amount when a plan is selected (best-effort default; admin
   724→  // can override). We don't have plan price data, so we just leave amount
   725→  // editable. Set the default amount to a round number based on sessionCount
   726→  // × 50000 as a convenience hint.
   727→  useEffect(() => {
   728→    if (packagePlanId && !amount) {
   729→      const plan = plans.find((p) => p.id === packagePlanId);
   730→      if (plan) {
   731→        setAmount(String(plan.sessionCount * 50000));
   732→      }
   733→    }
   734→  }, [packagePlanId, plans, amount]);
   735→
   736→  const handleSubmit = async () => {
   737→    if (!studentId) {
   738→      setError("Select a student.");
   739→      return;
   740→    }
   741→    if (!packagePlanId) {
   742→      setError("Select a package plan.");
   743→      return;
   744→    }
   745→    const amt = Number(amount);
   746→    if (!amt || amt <= 0) {
   747→      setError("Amount must be greater than zero.");
   748→      return;
   749→    }
   750→    if (!paymentDate) {
   751→      setError("Payment date is required.");
   752→      return;
   753→    }
   754→    if (!payerName.trim()) {
   755→      setError("Payer name is required.");
   756→      return;
   757→    }
   758→    setSubmitting(true);
   759→    setError(null);
   760→    try {
   761→      if (!idempotencyKeyRef.current) {
   762→        idempotencyKeyRef.current = crypto.randomUUID();
   763→      }
   764→      const body: Record<string, unknown> = {
   765→        studentId,
   766→        packagePlanId,
   767→        amount: amt,
   768→        paymentMethod,
   769→        paymentDate,
   770→        payerName: payerName.trim(),
   771→        idempotencyKey: idempotencyKeyRef.current,
   772→      };
   773→      if (referenceNumber.trim()) body.referenceNumber = referenceNumber.trim();
   774→      if (notes.trim()) body.notes = notes.trim();
   775→      const res = await adminFetch(`/api/admin/payments?academyId=${academyId}`, "POST", body);
   776→      if (res.ok || res.status === 201) {
   777→        onRecorded();
   778→        return;
   779→      }
   780→      const data = (await res.json()) as { error?: { message?: string } };
   781→      setError(data.error?.message ?? `Error ${res.status}`);
   782→    } catch {
   783→      setError("Network error");
   784→    } finally {
   785→      setSubmitting(false);
   786→    }
   787→  };
   788→
   789→  const handleClose = () => {
   790→    if (submitting) return;
   791→    onClose();
   792→  };
   793→
   794→  return (
   795→    <Dialog open={open} onOpenChange={(v) => { if (!v) handleClose(); }}>
   796→      <DialogContent className="sm:max-w-lg">
   797→        <DialogHeader>
   798→          <DialogTitle>Record Manual Payment</DialogTitle>
   799→          <DialogDescription>
   800→            Records a confirmed cash or bank-transfer payment and atomically purchases the package + creates entitlements.
   801→          </DialogDescription>
   802→        </DialogHeader>
   803→        <div className="space-y-4">
   804→          <div className="space-y-2">
   805→            <Label htmlFor="pay-student">Student *</Label>
   806→            <Select value={studentId} onValueChange={setStudentId} disabled={submitting}>
   807→              <SelectTrigger id="pay-student">
   808→                <SelectValue placeholder="Select student" />
   809→              </SelectTrigger>
   810→              <SelectContent>
   811→                {students.map((s) => (
   812→                  <SelectItem key={s.id} value={s.id}>
   813→                    {s.full_name}
   814→                  </SelectItem>
   815→                ))}
   816→              </SelectContent>
   817→            </Select>
   818→          </div>
   819→          <div className="space-y-2">
   820→            <Label htmlFor="pay-plan">Package Plan *</Label>
   821→            <Select value={packagePlanId} onValueChange={setPackagePlanId} disabled={submitting}>
   822→              <SelectTrigger id="pay-plan">
   823→                <SelectValue placeholder="Select plan" />
   824→              </SelectTrigger>
   825→              <SelectContent>
   826→                {plans.map((p) => (
   827→                  <SelectItem key={p.id} value={p.id}>
   828→                    {p.name} ({p.sessionCount} sessions)
   829→                  </SelectItem>
   830→                ))}
   831→              </SelectContent>
   832→            </Select>
   833→          </div>
   834→          <div className="grid grid-cols-2 gap-4">
   835→            <div className="space-y-2">
   836→              <Label htmlFor="pay-amount">Amount (IDR) *</Label>
   837→              <Input
   838→                id="pay-amount"
   839→                type="number"
   840→                min={1}
   841→                value={amount}
   842→                onChange={(e) => setAmount(e.target.value)}
   843→                disabled={submitting}
   844→                placeholder="500000"
   845→              />
   846→            </div>
   847→            <div className="space-y-2">
   848→              <Label htmlFor="pay-method">Method *</Label>
   849→              <Select value={paymentMethod} onValueChange={(v) => setPaymentMethod(v as "cash" | "bank_transfer")} disabled={submitting}>
   850→                <SelectTrigger id="pay-method">
   851→                  <SelectValue />
   852→                </SelectTrigger>
   853→                <SelectContent>
   854→                  <SelectItem value="cash">Cash</SelectItem>
   855→                  <SelectItem value="bank_transfer">Bank Transfer</SelectItem>
   856→                </SelectContent>
   857→              </Select>
   858→            </div>
   859→          </div>
   860→          <div className="grid grid-cols-2 gap-4">
   861→            <div className="space-y-2">
   862→              <Label htmlFor="pay-date">Payment Date *</Label>
   863→              <Input id="pay-date" type="date" value={paymentDate} onChange={(e) => setPaymentDate(e.target.value)} disabled={submitting} />
   864→            </div>
   865→            <div className="space-y-2">
   866→              <Label htmlFor="pay-ref">Reference #</Label>
   867→              <Input id="pay-ref" value={referenceNumber} onChange={(e) => setReferenceNumber(e.target.value)} disabled={submitting} placeholder="Optional" />
   868→            </div>
   869→          </div>
   870→          <div className="space-y-2">
   871→            <Label htmlFor="pay-payer">Payer Name *</Label>
   872→            <Input id="pay-payer" value={payerName} onChange={(e) => setPayerName(e.target.value)} disabled={submitting} placeholder="Parent / guardian name" />
   873→          </div>
   874→          <div className="space-y-2">
   875→            <Label htmlFor="pay-notes">Notes</Label>
   876→            <Textarea id="pay-notes" value={notes} onChange={(e) => setNotes(e.target.value)} disabled={submitting} placeholder="Optional" rows={2} />
   877→          </div>
   878→          {error && (
   879→            <div className="flex items-start gap-2 rounded-md border border-destructive/30 bg-destructive/5 p-3">
   880→              <AlertCircle className="h-4 w-4 text-destructive flex-shrink-0 mt-0.5" aria-hidden />
   881→              <p className="text-sm text-destructive">{error}</p>
   882→            </div>

   884→        </div>

   891→              <>
   892→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> Recording…

   895→              "Record Payment"
   896→            )}
   897→          </Button>
   898→        </DialogFooter>
   899→      </DialogContent>
   900→    </Dialog>
   901→  );
