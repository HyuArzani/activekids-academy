   138→function jakartaTodayYmd(): string {
   139→  const shifted = new Date(Date.now() + 7 * 60 * 60 * 1000);
   140→  const y = shifted.getUTCFullYear();
   141→  const m = String(shifted.getUTCMonth() + 1).padStart(2, "0");
   142→  const d = String(shifted.getUTCDate()).padStart(2, "0");
   143→  return `${y}-${m}-${d}`;
   144→}
   145→
   146→function jakartaMonthStart(): string {
   147→  const shifted = new Date(Date.now() + 7 * 60 * 60 * 1000);
   148→  const y = shifted.getUTCFullYear();
   149→  const m = String(shifted.getUTCMonth() + 1).padStart(2, "0");
   150→  return `${y}-${m}-01`;
   151→}

   153→function formatRupiah(amountStr: string): string {
   154→  const n = Number(amountStr);
   155→  if (Number.isNaN(n)) return `Rp ${amountStr}`;
   156→  return `Rp ${n.toLocaleString("id-ID", { maximumFractionDigits: 0 })}`;
   157→}
   158→
   159→function formatDate(ymd: string | null): string {
   160→  if (!ymd) return "—";
   161→  const m = /^(\d{4})-(\d{2})-(\d{2})/.exec(ymd);
   162→  if (!m) return ymd;
   163→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   164→  const monthIdx = parseInt(m[2]!, 10) - 1;
   165→  return `${m[3]} ${months[monthIdx] ?? "???"} ${m[1]}`;
   166→}

   235→function StudentRosterReport({ academyId }: { academyId: string }) {
   236→  const [state, setState] = useState<{ status: LoadingState; data?: StudentRosterReport; error?: string }>({ status: "loading" });
   237→  const [search, setSearch] = useState("");
   238→  const academyIdRef = useRef(academyId);
   239→  useEffect(() => {
   240→    academyIdRef.current = academyId;
   241→  }, [academyId]);
   242→
   243→  // Request-ordering token: incremented on each fetchReport/effect-fetch
   244→  // call. Before applying a response, check if this token still matches
   245→  // the latest value. If a newer request has started, the stale response
   246→  // is discarded. This prevents an older (slower) request from
   247→  // overwriting newer data — e.g., a post-mutation refetch racing a
   248→  // date-range change, or a manual Refresh racing the deps-change fetch.
   249→  // Mirrors the RegistrationsManagement pattern (Phase B3.3.2).
   250→  const requestIdRef = useRef(0);
   251→
   252→  const fetchReport = useCallback(async () => {
   253→    const aid = academyId;
   254→    const myRequestId = ++requestIdRef.current;
   255→    setState({ status: "loading" });
   256→    try {
   257→      const res = await fetch(`/api/admin/reports/students?academyId=${aid}`, { credentials: "same-origin" });
   258→      if (myRequestId !== requestIdRef.current) return;
   259→      if (aid !== academyIdRef.current) return;
   260→      if (!res.ok) {
   261→        const data = (await res.json()) as { error?: { message?: string } };
   262→        setState({ status: "error", error: data.error?.message ?? `Error ${res.status}` });
   263→        return;
   264→      }
   265→      const data = (await res.json()) as StudentRosterReport;
   266→      setState({ status: "success", data });
   267→    } catch {
   268→      if (myRequestId !== requestIdRef.current) return;
   269→      if (aid !== academyIdRef.current) return;
   270→      setState({ status: "error", error: "Network error" });
   271→    }
   272→  }, [academyId]);
   273→
   274→  useEffect(() => {
   275→    if (!academyId) return;
   276→    let cancelled = false;
   277→    const myRequestId = ++requestIdRef.current;
   278→    void (async () => {
   279→      try {
   280→        const res = await fetch(`/api/admin/reports/students?academyId=${academyId}`, { credentials: "same-origin" });
   281→        if (cancelled || myRequestId !== requestIdRef.current) return;
   282→        if (!res.ok) {
   283→          const data = (await res.json()) as { error?: { message?: string } };
   284→          setState({ status: "error", error: data.error?.message ?? `Error ${res.status}` });
   285→          return;
   286→        }
   287→        const data = (await res.json()) as StudentRosterReport;
   288→        if (cancelled || myRequestId !== requestIdRef.current) return;
   289→        setState({ status: "success", data });
   290→      } catch {
   291→        if (cancelled || myRequestId !== requestIdRef.current) return;
   292→        setState({ status: "error", error: "Network error" });
   293→      }
   294→    })();
   295→    return () => { cancelled = true; };
   296→  }, [academyId]);
   297→
   298→  const filteredStudents = state.data?.students.filter((s) =>
   299→    s.fullName.toLowerCase().includes(search.toLowerCase()),
   300→  ) ?? [];
   301→
   302→  if (state.status === "loading") return <LoadingCard text="Loading student roster…" />;
   303→  if (state.status === "error") return <ErrorCard message={state.error!} onRetry={fetchReport} />;
   304→
   305→  return (
   306→    <div className="space-y-4">
   307→      {/* Summary cards */}
   308→      <div className="grid gap-3 sm:grid-cols-3">
   309→        <SummaryCard label="Total Students" value={state.data!.totalStudents} icon={Users} color="primary" />
   310→        <SummaryCard label="Active" value={state.data!.activeStudents} icon={CheckCircle2} color="emerald" />
   311→        <SummaryCard label="Archived" value={state.data!.archivedStudents} icon={XCircle} color="muted" />
   312→      </div>
   313→
   314→      {/* Search + refresh + export */}
   315→      <div className="flex items-center gap-2">
   316→        <Input
   317→          placeholder="Search students…"
   318→          value={search}
   319→          onChange={(e) => setSearch(e.target.value)}
   320→          className="max-w-xs"
   321→        />
   322→        <Button variant="outline" size="sm" onClick={fetchReport} className="gap-2">
   323→          <RefreshCw className="h-4 w-4" aria-hidden /> Refresh
   324→        </Button>
   325→        <Button
   326→          variant="outline"
   327→          size="sm"
   328→          onClick={() => downloadStudentRosterCSV(filteredStudents)}
   329→          disabled={filteredStudents.length === 0}
   330→          className="gap-2"
   331→        >
   332→          <Download className="h-4 w-4" aria-hidden /> CSV
   333→        </Button>
   334→      </div>
   335→
   336→      {/* Table */}
   337→      {filteredStudents.length === 0 ? (
   338→        <Card>
   339→          <CardContent className="p-8 text-center">
   340→            <Users className="mx-auto h-10 w-10 text-muted-foreground" aria-hidden />
   341→            <p className="mt-3 text-sm font-medium">{search ? "No matching students" : "No students yet"}</p>
   342→          </CardContent>
   343→        </Card>
   344→      ) : (
   345→        <Card>
   346→          <CardContent className="p-0">
   347→            <div className="max-h-[600px] overflow-auto">
   348→              <Table>
   349→                <TableHeader>
   350→                  <TableRow>
   351→                    <TableHead>Student</TableHead>
   352→                    <TableHead>Class</TableHead>
   353→                    <TableHead className="text-center">Packages</TableHead>
   354→                    <TableHead className="text-center">Active</TableHead>
   355→                    <TableHead className="text-center">Completed</TableHead>
   356→                    <TableHead className="text-center">Makeup</TableHead>
   357→                    <TableHead>Last Attendance</TableHead>
   358→                  </TableRow>
   359→                </TableHeader>
   360→                <TableBody>
   361→                  {filteredStudents.map((s) => (
   362→                    <TableRow key={s.studentId}>
   363→                      <TableCell>
   364→                        <div className="flex items-center gap-2">
   365→                          <div className="flex h-7 w-7 items-center justify-center rounded-full bg-primary/10 text-xs font-semibold text-primary">
   366→                            {s.fullName.charAt(0).toUpperCase()}
   367→                          </div>
   368→                          <div>
   369→                            <p className="font-medium">{s.fullName}</p>
   370→                            {s.archivedAt && <Badge variant="outline" className="text-xs">Archived</Badge>}
   371→                          </div>
   372→                        </div>
   373→                      </TableCell>
   374→                      <TableCell>{s.currentClassName ?? "—"}</TableCell>
   375→                      <TableCell className="text-center">{s.packageCount}</TableCell>
   376→                      <TableCell className="text-center">
   377→                        {s.activeEntitlements > 0 ? (
   378→                          <Badge variant="secondary">{s.activeEntitlements}</Badge>
   379→                        ) : "—"}
   380→                      </TableCell>
   381→                      <TableCell className="text-center">
   382→                        {s.completedEntitlements > 0 ? (
   383→                          <Badge variant="secondary">{s.completedEntitlements}</Badge>
   384→                        ) : "—"}
   385→                      </TableCell>
   386→                      <TableCell className="text-center">
   387→                        {s.awaitingMakeupEntitlements > 0 ? (
   388→                          <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-400">
   389→                            {s.awaitingMakeupEntitlements}
   390→                          </Badge>
   391→                        ) : "—"}
   392→                      </TableCell>
   393→                      <TableCell>
   394→                        {s.lastAttendedDate ? (
   395→                          <div className="flex items-center gap-1.5">
   396→                            <span className="text-xs">{formatDate(s.lastAttendedDate)}</span>
   397→                            {s.lastAttendedStatus && STATUS_META[s.lastAttendedStatus] && (
   398→                              <Badge variant="outline" className={`text-xs ${STATUS_META[s.lastAttendedStatus]!.color}`}>
   399→                                {STATUS_META[s.lastAttendedStatus]!.label}
   400→                              </Badge>
   401→                            )}
   402→                          </div>
   403→                        ) : "—"}
   404→                      </TableCell>
   405→                    </TableRow>
   406→                  ))}
   407→                </TableBody>
   408→              </Table>
   409→            </div>
   410→          </CardContent>
   411→        </Card>
   412→      )}
   413→    </div>
   414→  );
   415→}

   421→function AttendanceReport({ academyId }: { academyId: string }) {
   422→  const [students, setStudents] = useState<Student[]>([]);
   423→  const [selectedStudentId, setSelectedStudentId] = useState("");
   424→  const [report, setReport] = useState<PerStudentAttendanceReport | null>(null);
   425→  const [loading, setLoading] = useState(false);
   426→  const [error, setError] = useState<string | null>(null);
   427→  const academyIdRef = useRef(academyId);
   428→  useEffect(() => {
   429→    academyIdRef.current = academyId;
   430→  }, [academyId]);
   431→
   432→  // Request-ordering token: incremented on each fetchReport/effect-fetch
   433→  // call. Before applying a response, check if this token still matches
   434→  // the latest value. If a newer request has started, the stale response
   435→  // is discarded. This prevents an older (slower) request from
   436→  // overwriting newer data — e.g., a post-mutation refetch racing a
   437→  // date-range change, or a manual Refresh racing the deps-change fetch.
   438→  // Mirrors the RegistrationsManagement pattern (Phase B3.3.2).
   439→  //
   440→  // Note: this component's report-fetch useEffect (below) delegates to
   441→  // fetchReport directly (`void fetchReport()`) rather than duplicating
   442→  // the fetch logic in an IIFE — so only the useCallback needs the guard.
   443→  // The students-list useEffect IIFE below fetches a DIFFERENT resource
   444→  // (/api/admin/students) and does not race with fetchReport, so it is
   445→  // left unchanged.
   446→  const requestIdRef = useRef(0);
   447→
   448→  // Load students list.
   449→  useEffect(() => {
   450→    if (!academyId) return;
   451→    let cancelled = false;
   452→    void (async () => {
   453→      try {
   454→        const res = await fetch(`/api/admin/students?academyId=${academyId}`, { credentials: "same-origin" });
   455→        if (cancelled) return;
   456→        const data = (await res.json()) as { students?: Student[] };
   457→        if (cancelled) return;
   458→        setStudents(data.students ?? []);
   459→      } catch {
   460→        // ignore
   461→      }
   462→    })();
   463→    return () => { cancelled = true; };
   464→  }, [academyId]);
   465→
   466→  const fetchReport = useCallback(async () => {
   467→    if (!selectedStudentId) return;
   468→    const aid = academyId;
   469→    const myRequestId = ++requestIdRef.current;
   470→    setLoading(true);
   471→    setError(null);
   472→    try {
   473→      const res = await fetch(
   474→        `/api/admin/reports/attendance?academyId=${aid}&studentId=${selectedStudentId}`,
   475→        { credentials: "same-origin" },
   476→      );
   477→      if (myRequestId !== requestIdRef.current) return;
   478→      if (aid !== academyIdRef.current) return;
   479→      if (!res.ok) {
   480→        const data = (await res.json()) as { error?: { message?: string } };
   481→        setError(data.error?.message ?? `Error ${res.status}`);
   482→        return;
   483→      }
   484→      const data = (await res.json()) as PerStudentAttendanceReport;
   485→      setReport(data);
   486→    } catch {
   487→      if (myRequestId !== requestIdRef.current) return;
   488→      if (aid !== academyIdRef.current) return;
   489→      setError("Network error");
   490→    } finally {
   491→      setLoading(false);
   492→    }
   493→  }, [academyId, selectedStudentId]);
   494→
   495→  useEffect(() => {
   496→    if (selectedStudentId) void fetchReport();
   497→    else setReport(null);
   498→  }, [fetchReport, selectedStudentId]);
   499→
   500→  return (
   501→    <div className="space-y-4">
   502→      {/* Student selector */}
   503→      <Card>
   504→        <CardContent className="p-4">
   505→          <div className="flex items-end gap-3">
   506→            <div className="flex-1 space-y-2">
   507→              <Label htmlFor="report-student">Select Student</Label>
   508→              <Select value={selectedStudentId} onValueChange={setSelectedStudentId}>
   509→                <SelectTrigger id="report-student">
   510→                  <SelectValue placeholder="Choose a student…" />
   511→                </SelectTrigger>
   512→                <SelectContent>
   513→                  {students.map((s) => (
   514→                    <SelectItem key={s.id} value={s.id}>
   515→                      {s.full_name}
   516→                    </SelectItem>
   517→                  ))}
   518→                </SelectContent>
   519→              </Select>
   520→            </div>
   521→            <Button variant="outline" size="sm" onClick={fetchReport} disabled={!selectedStudentId || loading} className="gap-2">
   522→              {loading ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <RefreshCw className="h-4 w-4" aria-hidden />}
   523→              Refresh
   524→            </Button>
   525→          </div>
   526→        </CardContent>
   527→      </Card>
   528→
   529→      {!selectedStudentId && (
   530→        <Card>
   531→          <CardContent className="p-8 text-center">
   532→            <ClipboardCheck className="mx-auto h-10 w-10 text-muted-foreground" aria-hidden />
   533→            <p className="mt-3 text-sm font-medium">Select a student to view their attendance report</p>
   534→          </CardContent>
   535→        </Card>
   536→      )}
   537→
   538→      {error && <ErrorCard message={error} onRetry={fetchReport} />}
   539→
   540→      {report && (
   541→        <>
   542→          {/* Summary cards */}
   543→          <div className="grid gap-3 sm:grid-cols-3 lg:grid-cols-6">
   544→            <SummaryCard label="Total" value={report.summary.total} icon={Calendar} color="primary" />
   545→            <SummaryCard label="Present" value={report.summary.present} icon={CheckCircle2} color="emerald" />

   605→  const [report, setReport] = useState<MonthlyFinanceReport | null>(null);
   606→  const [loading, setLoading] = useState(false);
   607→  const [error, setError] = useState<string | null>(null);
   608→  const academyIdRef = useRef(academyId);
   609→  useEffect(() => {
   610→    academyIdRef.current = academyId;
   611→  }, [academyId]);
   612→
   613→  // Request-ordering token: incremented on each fetchReport/effect-fetch
   614→  // call. Before applying a response, check if this token still matches
   615→  // the latest value. If a newer request has started, the stale response
   616→  // is discarded. This prevents an older (slower) request from
   617→  // overwriting newer data — e.g., a post-mutation refetch racing a
   618→  // date-range change, or a manual Refresh racing the deps-change fetch.
   619→  // Mirrors the RegistrationsManagement pattern (Phase B3.3.2).
   620→  const requestIdRef = useRef(0);
   621→
   622→  const fetchReport = useCallback(async () => {
   623→    const aid = academyId;
   624→    const myRequestId = ++requestIdRef.current;
   625→    setLoading(true);
   626→    setError(null);
   627→    try {
   628→      const res = await fetch(
   629→        `/api/admin/reports/finance?academyId=${aid}&fromDate=${fromDate}&toDate=${toDate}`,
   630→        { credentials: "same-origin" },
   631→      );
   632→      if (myRequestId !== requestIdRef.current) return;
   633→      if (aid !== academyIdRef.current) return;
   634→      if (!res.ok) {
   635→        const data = (await res.json()) as { error?: { message?: string } };
   636→        setError(data.error?.message ?? `Error ${res.status}`);
   637→        return;
   638→      }
   639→      const data = (await res.json()) as MonthlyFinanceReport;
   640→      setReport(data);
   641→    } catch {
   642→      if (myRequestId !== requestIdRef.current) return;
   643→      if (aid !== academyIdRef.current) return;
   644→      setError("Network error");
   645→    } finally {
   646→      setLoading(false);
   647→    }
   648→  }, [academyId, fromDate, toDate]);
   649→
   650→  useEffect(() => {
   651→    if (!academyId || !fromDate || !toDate) return;
   652→    let cancelled = false;
   653→    const myRequestId = ++requestIdRef.current;
   654→    void (async () => {
   655→      setLoading(true);
   656→      setError(null);
   657→      try {
   658→        const res = await fetch(
   659→          `/api/admin/reports/finance?academyId=${academyId}&fromDate=${fromDate}&toDate=${toDate}`,

   670→        setReport(data);
   671→      } catch {
   672→        if (cancelled || myRequestId !== requestIdRef.current) return;
   673→        setError("Network error");
   674→      } finally {
   675→        if (!cancelled) setLoading(false);
   676→      }

   923→function escapeCSV(value: string | number | null | undefined): string {
   924→  if (value === null || value === undefined) return "";
   925→  const s = String(value);
   926→  // Quote if contains comma, quote, or newline.
   927→  if (s.includes(",") || s.includes('"') || s.includes("\n")) {
   928→    return `"${s.replace(/"/g, '""')}"`;
   929→  }
   930→  return s;
   931→}
   932→
   933→function downloadCSV(filename: string, rows: string[][]): void {
   934→  const csv = rows.map((r) => r.map(escapeCSV).join(",")).join("\n");
   935→  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   936→  const url = URL.createObjectURL(blob);
   937→  const link = document.createElement("a");
   938→  link.href = url;
   939→  link.download = filename;
   940→  document.body.appendChild(link);
   941→  link.click();
   942→  document.body.removeChild(link);
   943→  URL.revokeObjectURL(url);
   944→}

   961→function downloadFinanceCSV(payments: FinanceReportPayment[], expenses: FinanceReportExpense[], fromDate: string, toDate: string): void {
   962→  // Two sections: payments then expenses, separated by a blank row.
   963→  const paymentHeaders = ["Type", "Date", "Student/Category", "Method/Note", "Package", "Status", "Amount"];
   964→  const paymentRows = payments.map((p) => [
   965→    "Payment",

   969→    p.packageName ?? "",
   970→    p.status,

   974→  const expenseRows = expenses.map((e) => [
   975→    "Expense",
   976→    e.incurredOn,
   977→    e.category,
   978→    e.note ?? "",
   979→    "",
   980→    "",
   981→    e.amount,
   982→  ]);
   983→  downloadCSV(`finance-report-${fromDate}-to-${toDate}.csv`, [
   984→    paymentHeaders,
   985→    ...paymentRows,
   986→    [], // blank separator row
   987→    expenseHeaders,
