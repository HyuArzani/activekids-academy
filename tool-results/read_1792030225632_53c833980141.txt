   138→   138→function jakartaTodayYmd(): string {
   139→   139→  const shifted = new Date(Date.now() + 7 * 60 * 60 * 1000);
   140→   140→  const y = shifted.getUTCFullYear();
   141→   141→  const m = String(shifted.getUTCMonth() + 1).padStart(2, "0");
   142→   142→  const d = String(shifted.getUTCDate()).padStart(2, "0");
   143→   143→  return `${y}-${m}-${d}`;
   144→   144→}

   153→   153→function formatRupiah(amountStr: string): string {
   154→   154→  const n = Number(amountStr);
   155→   155→  if (Number.isNaN(n)) return `Rp ${amountStr}`;
   156→   156→  return `Rp ${n.toLocaleString("id-ID", { maximumFractionDigits: 0 })}`;
   157→   157→}
   158→   158→
   159→   159→function formatDate(ymd: string | null): string {
   160→   160→  if (!ymd) return "—";
   161→   161→  const m = /^(\d{4})-(\d{2})-(\d{2})/.exec(ymd);
   162→   162→  if (!m) return ymd;
   163→   163→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   164→   164→  const monthIdx = parseInt(m[2]!, 10) - 1;
   165→   165→  return `${m[3]} ${months[monthIdx] ?? "???"} ${m[1]}`;
   166→   166→}

   235→   235→function StudentRosterReport({ academyId }: { academyId: string }) {
   236→   236→  const [state, setState] = useState<{ status: LoadingState; data?: StudentRosterReport; error?: string }>({ status: "loading" });
   237→   237→  const [search, setSearch] = useState("");
   238→   238→  const academyIdRef = useRef(academyId);
   239→   239→  useEffect(() => {
   240→   240→    academyIdRef.current = academyId;
   241→   241→  }, [academyId]);
   242→   242→
   243→   243→  // Request-ordering token: incremented on each fetchReport/effect-fetch
   244→   244→  // call. Before applying a response, check if this token still matches
   245→   245→  // the latest value. If a newer request has started, the stale response
   246→   246→  // is discarded. This prevents an older (slower) request from
   247→   247→  // overwriting newer data — e.g., a post-mutation refetch racing a
   248→   248→  // date-range change, or a manual Refresh racing the deps-change fetch.
   249→   249→  // Mirrors the RegistrationsManagement pattern (Phase B3.3.2).
   250→   250→  const requestIdRef = useRef(0);
   251→   251→
   252→   252→  const fetchReport = useCallback(async () => {
   253→   253→    const aid = academyId;
   254→   254→    const myRequestId = ++requestIdRef.current;
   255→   255→    setState({ status: "loading" });
   256→   256→    try {
   257→   257→      const res = await fetch(`/api/admin/reports/students?academyId=${aid}`, { credentials: "same-origin" });
   258→   258→      if (myRequestId !== requestIdRef.current) return;
   259→   259→      if (aid !== academyIdRef.current) return;
   260→   260→      if (!res.ok) {
   261→   261→        const data = (await res.json()) as { error?: { message?: string } };
   262→   262→        setState({ status: "error", error: data.error?.message ?? `Error ${res.status}` });
   263→   263→        return;
   264→   264→      }
   265→   265→      const data = (await res.json()) as StudentRosterReport;
   266→   266→      setState({ status: "success", data });
   267→   267→    } catch {
   268→   268→      if (myRequestId !== requestIdRef.current) return;
   269→   269→      if (aid !== academyIdRef.current) return;
   270→   270→      setState({ status: "error", error: "Network error" });
   271→   271→    }
   272→   272→  }, [academyId]);
   273→   273→
   274→   274→  useEffect(() => {
   275→   275→    if (!academyId) return;
   276→   276→    let cancelled = false;
   277→   277→    const myRequestId = ++requestIdRef.current;
   278→   278→    void (async () => {
   279→   279→      try {
   280→   280→        const res = await fetch(`/api/admin/reports/students?academyId=${academyId}`, { credentials: "same-origin" });
   281→   281→        if (cancelled || myRequestId !== requestIdRef.current) return;
   282→   282→        if (!res.ok) {
   283→   283→          const data = (await res.json()) as { error?: { message?: string } };
   284→   284→          setState({ status: "error", error: data.error?.message ?? `Error ${res.status}` });
   285→   285→          return;
   286→   286→        }
   287→   287→        const data = (await res.json()) as StudentRosterReport;
   288→   288→        if (cancelled || myRequestId !== requestIdRef.current) return;
   289→   289→        setState({ status: "success", data });
   290→   290→      } catch {
   291→   291→        if (cancelled || myRequestId !== requestIdRef.current) return;
   292→   292→        setState({ status: "error", error: "Network error" });
   293→   293→      }
   294→   294→    })();
   295→   295→    return () => { cancelled = true; };
   296→   296→  }, [academyId]);
   297→   297→
   298→   298→  const filteredStudents = state.data?.students.filter((s) =>
   299→   299→    s.fullName.toLowerCase().includes(search.toLowerCase()),
   300→   300→  ) ?? [];
   301→   301→
   302→   302→  if (state.status === "loading") return <LoadingCard text="Loading student roster…" />;
   303→   303→  if (state.status === "error") return <ErrorCard message={state.error!} onRetry={fetchReport} />;
   304→   304→
   305→   305→  return (
   306→   306→    <div className="space-y-4">
   307→   307→      {/* Summary cards */}
   308→   308→      <div className="grid gap-3 sm:grid-cols-3">
   309→   309→        <SummaryCard label="Total Students" value={state.data!.totalStudents} icon={Users} color="primary" />
   310→   310→        <SummaryCard label="Active" value={state.data!.activeStudents} icon={CheckCircle2} color="emerald" />
   311→   311→        <SummaryCard label="Archived" value={state.data!.archivedStudents} icon={XCircle} color="muted" />
   312→   312→      </div>
   313→   313→
   314→   314→      {/* Search + refresh + export */}
   315→   315→      <div className="flex items-center gap-2">
   316→   316→        <Input
   317→   317→          placeholder="Search students…"
   318→   318→          value={search}
   319→   319→          onChange={(e) => setSearch(e.target.value)}
   320→   320→          className="max-w-xs"
   321→   321→        />
   322→   322→        <Button variant="outline" size="sm" onClick={fetchReport} className="gap-2">
   323→   323→          <RefreshCw className="h-4 w-4" aria-hidden /> Refresh
   324→   324→        </Button>
   325→   325→        <Button
   326→   326→          variant="outline"
   327→   327→          size="sm"
   328→   328→          onClick={() => downloadStudentRosterCSV(filteredStudents)}
   329→   329→          disabled={filteredStudents.length === 0}
   330→   330→          className="gap-2"
   331→   331→        >
   332→   332→          <Download className="h-4 w-4" aria-hidden /> CSV
   333→   333→        </Button>
   334→   334→      </div>
   335→   335→
   336→   336→      {/* Table */}
   337→   337→      {filteredStudents.length === 0 ? (
   338→   338→        <Card>
   339→   339→          <CardContent className="p-8 text-center">
   340→   340→            <Users className="mx-auto h-10 w-10 text-muted-foreground" aria-hidden />
   341→   341→            <p className="mt-3 text-sm font-medium">{search ? "No matching students" : "No students yet"}</p>
   342→   342→          </CardContent>
   343→   343→        </Card>
   344→   344→      ) : (
   345→   345→        <Card>
   346→   346→          <CardContent className="p-0">
   347→   347→            <div className="max-h-[600px] overflow-auto">
   348→   348→              <Table>
   349→   349→                <TableHeader>
   350→   350→                  <TableRow>
   351→   351→                    <TableHead>Student</TableHead>
   352→   352→                    <TableHead>Class</TableHead>
   353→   353→                    <TableHead className="text-center">Packages</TableHead>
   354→   354→                    <TableHead className="text-center">Active</TableHead>
   355→   355→                    <TableHead className="text-center">Completed</TableHead>
   356→   356→                    <TableHead className="text-center">Makeup</TableHead>
   357→   357→                    <TableHead>Last Attendance</TableHead>
   358→   358→                  </TableRow>
   359→   359→                </TableHeader>

   399→   399→                                {STATUS_META[s.lastAttendedStatus]!.label}
   400→   400→                              </Badge>
   401→   401→                            )}
   402→   402→                          </div>
   403→   403→                        ) : "—"}
   404→   404→                      </TableCell>
   405→   405→                    </TableRow>
   406→   406→                  ))}
   407→   407→                </TableBody>
   408→   408→              </Table>
   409→   409→            </div>
   410→   410→          </CardContent>
   411→   411→        </Card>
   412→   412→      )}
   413→   413→    </div>
   414→   414→  );

   439→   439→  //

   923→   923→function escapeCSV(value: string | number | null | undefined): string {
   924→   924→  if (value === null || value === undefined) return "";
   925→   925→  const s = String(value);
   926→   926→  // Quote if contains comma, quote, or newline.
   927→   927→  if (s.includes(",") || s.includes('"') || s.includes("\n")) {
   928→   928→    return `"${s.replace(/"/g, '""')}"`;
   929→   929→  }
   930→   930→  return s;
   931→   931→}
   932→   932→
   933→   933→function downloadCSV(filename: string, rows: string[][]): void {
   934→   934→  const csv = rows.map((r) => r.map(escapeCSV).join(",")).join("\n");
   935→   935→  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   936→   936→  const url = URL.createObjectURL(blob);
   937→   937→  const link = document.createElement("a");
   938→   938→  link.href = url;
   939→   939→  link.download = filename;
   940→   940→  document.body.appendChild(link);
   941→   941→  link.click();
   942→   942→  document.body.removeChild(link);
   943→   943→  URL.revokeObjectURL(url);
   944→   944→}
